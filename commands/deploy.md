## Step 2: Generate Script

Write `_keeper/daemon.sh`.

### Script template

```bash
#!/usr/bin/env bash
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
# 🔒 Keeper Daemon — generated by /keeper:deploy
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
#
# Lenses:  {comma-separated list}
# Hours:   {start}-{end} {timezone}
# Cadence: every {N}h
#
# Usage:
#   ./_keeper/daemon.sh start    Start daemon in tmux
#   ./_keeper/daemon.sh stop     Stop daemon
#   ./_keeper/daemon.sh status   Check if running
#   ./_keeper/daemon.sh logs     Tail daemon output
#
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

set -euo pipefail

SESSION_NAME="keeper-{project-name}"
PROJECT_DIR="{absolute project path}"
PLUGIN_DIR="{absolute plugin path}"
CADENCE_SECONDS={cadence in seconds}
WORK_START="{HH:MM}"
WORK_END="{HH:MM}"
TIMEZONE="{timezone}"

# ── Helpers ─────────────────────────────────

is_running() {
  tmux has-session -t "$SESSION_NAME" 2>/dev/null
}

in_working_hours() {
  local current_hour
  current_hour=$(TZ="$TIMEZONE" date +%H%M)
  local start="${WORK_START/:/}"
  local end="${WORK_END/:/}"
  [[ "$current_hour" -ge "$start" && "$current_hour" -lt "$end" ]]
}

# ── Commands ────────────────────────────────

cmd_start() {
  if is_running; then
    echo "🔒 Keeper daemon already running."
    echo "   Attach: tmux attach -t $SESSION_NAME"
    exit 0
  fi

  if ! in_working_hours; then
    echo "🔒 Outside working hours ($WORK_START-$WORK_END $TIMEZONE)."
    echo "   Starting anyway — keeper will sleep-consolidate and pause."
  fi

  tmux new-session -d -s "$SESSION_NAME" -c "$PROJECT_DIR"

  # Run keeper in a loop with cadence pauses
  tmux send-keys -t "$SESSION_NAME" "while true; do \\
    claude --plugin-dir \"$PLUGIN_DIR\" -p '/keeper:run --max-iterations 15'; \\
    echo '🔒 Cycle complete. Sleeping ${CADENCE_SECONDS}s...'; \\
    sleep $CADENCE_SECONDS; \\
  done" Enter

  echo "🔒 Keeper daemon started."
  echo ""
  echo "   Session:  $SESSION_NAME"
  echo "   Project:  $PROJECT_DIR"
  echo "   Cadence:  every $((CADENCE_SECONDS / 3600))h"
  echo "   Hours:    $WORK_START-$WORK_END $TIMEZONE"
  echo ""
  echo "   Attach:   tmux attach -t $SESSION_NAME"
  echo "   Stop:     $0 stop"
  echo "   Logs:     $0 logs"
}

cmd_stop() {
  if ! is_running; then
    echo "🔒 Keeper daemon is not running."
    exit 0
  fi

  tmux kill-session -t "$SESSION_NAME"
  echo "🔒 Keeper daemon stopped."
}

cmd_status() {
  if is_running; then
    echo "🔒 Keeper daemon is running."
    echo "   Session: $SESSION_NAME"
    echo ""
    echo "   Last output:"
    tmux capture-pane -t "$SESSION_NAME" -p | tail -5
  else
    echo "🔒 Keeper daemon is not running."
  fi
}

cmd_logs() {
  if ! is_running; then
    echo "🔒 Keeper daemon is not running."
    exit 1
  fi

  echo "🔒 Attaching to keeper output (Ctrl+C to detach)..."
  tmux attach -t "$SESSION_NAME"
}

# ── Main ────────────────────────────────────

case "${1:-start}" in
  start)  cmd_start ;;
  stop)   cmd_stop ;;
  status) cmd_status ;;
  logs)   cmd_logs ;;
  *)
    echo "Usage: $0 {start|stop|status|logs}"
    exit 1
    ;;
esac
```

## HARD RULES

1. **Always use AskUserQuestion.** This is an interactive command — never guess settings.
2. **Absolute paths in script.** The script runs from anywhere; all paths must be absolute.
3. **Offer .gitignore.** The script contains local paths — it shouldn't be committed.
4. **Detect, don't assume.** Auto-detect plugin path, project path, existing config.
5. **Idempotent.** Running deploy twice should detect existing script and ask what to do.
6. **No cron.** Use tmux loop with sleep, not crontab entries. Simpler, more visible, easier to debug.
